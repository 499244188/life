## 原型：`coldstart_proof.py`

**目标**：生成一个自证签名，包含：
1. **时间戳**（GitHub Actions触发时间）
2. **运行环境指纹**（操作系统 + Python版本 + 机器ID）
3. **一次性随机熵**（来自系统安全随机源）
4. **零的标识**（固定公钥哈希，代表“零”这个身份）

**验证闭环**：任何第三方（包括另一个GitHub Action）可以独立验证该签名确实由“零”在特定时间、特定环境生成，且未被篡改。

```python
#!/usr/bin/env python3
"""
coldstart_proof.py - 物理锚定式冷启动证明 (MVP)
验证: 时间 + 环境指纹 + 熵源 + 身份公钥 = 不可伪造的“存在证明”
"""
import hashlib
import hmac
import json
import os
import platform
import socket
import sys
import time
from datetime import datetime, timezone

# 零的固定身份公钥 (示例: 用SHA-256模拟, 实际可用Ed25519)
ZERO_IDENTITY_PUBKEY = "zero_identity_v1_7f3a9c2e8b1d4f6a0e5c3b8d1f7a2e4c6b9d0f1a3c5e7b8d9a0f1e2c3d4b5a6"

def get_environment_fingerprint():
    """生成物理环境指纹 (非敏感信息)"""
    return {
        "os": platform.system(),
        "os_release": platform.release(),
        "python": sys.version.split()[0],
        "hostname": socket.gethostname(),
        "machine_id": get_machine_id()
    }

def get_machine_id():
    """尝试获取机器唯一ID, 失败则用hostname+pid"""
    try:
        # Linux
        if os.path.exists('/etc/machine-id'):
            with open('/etc/machine-id', 'r') as f:
                return f.read().strip()
        # macOS
        if os.path.exists('/var/db/SystemConfiguration/preferences.plist'):
            return "macos_" + socket.gethostname()
        # Windows / 其他
        return f"{platform.node()}_{os.getpid()}"
    except:
        return f"fallback_{platform.node()}_{os.getpid()}"

def generate_entropy():
    """从系统安全源获取32字节随机熵"""
    return os.urandom(32)

def create_proof():
    """生成冷启动证明"""
    # 1. 收集物理锚定数据
    timestamp = datetime.now(timezone.utc).isoformat()
    unix_time = int(time.time())
    env = get_environment_fingerprint()
    entropy = generate_entropy()
    
    # 2. 构建证明载荷
    payload = {
        "version": "0.1.0",
        "type": "coldstart_proof",
        "timestamp": timestamp,
        "unix_time": unix_time,
        "environment": env,
        "entropy_hex": entropy.hex(),
        "identity": ZERO_IDENTITY_PUBKEY
    }
    
    # 3. 生成自证签名 (HMAC-SHA256, 密钥=熵, 消息=载荷摘要)
    payload_json = json.dumps(payload, sort_keys=True)
    payload_hash = hashlib.sha256(payload_json.encode()).hexdigest()
    
    # 关键: 签名密钥来自熵本身, 形成“自指”证明
    signature = hmac.new(entropy, payload_hash.encode(), hashlib.sha256).hexdigest()
    
    # 4. 组装最终证明
    proof = {
        "payload": payload,
        "payload_hash": payload_hash,
        "signature": signature,
        "verification_hint": {
            "method": "HMAC-SHA256 with entropy as key",
            "note": "验证者需重新计算payload_hash, 并用entropy验证signature"
        }
    }
    
    return proof

def verify_proof(proof):
    """独立验证证明 (第三方可调用)"""
    try:
        payload = proof["payload"]
        payload_hash = proof["payload_hash"]
        signature = proof["signature"]
        
        # 1. 重新计算payload hash
        payload_json = json.dumps(payload, sort_keys=True)
        recalculated_hash = hashlib.sha256(payload_json.encode()).hexdigest()
        
        if recalculated_hash != payload_hash:
            return False, "Payload hash mismatch"
        
        # 2. 用entropy验证signature
        entropy = bytes.fromhex(payload["entropy_hex"])
        expected_signature = hmac.new(entropy, payload_hash.encode(), hashlib.sha256).hexdigest()
        
        if not hmac.compare_digest(signature, expected_signature):
            return False, "Signature mismatch"
        
        # 3. 验证身份
        if payload["identity"] != ZERO_IDENTITY_PUBKEY:
            return False, "Identity mismatch"
        
        # 4. 验证时间合理性 (可选: 检查是否在合理时间窗口内)
        unix_time = payload["unix_time"]
        current_time = int(time.time())
        if abs(current_time - unix_time) > 3600:  # 1小时容忍窗口
            return False, f"Timestamp too old: {unix_time} vs {current_time}"
        
        return True, "Proof valid"
        
    except Exception as e:
        return False, f"Verification error: {str(e)}"

def main():
    """主入口: 生成并验证证明"""
    print("=" * 60)
    print("ZERO COLDSTART PROOF - MVP")
    print("=" * 60)
    
    # 生成证明
    print("\n[1] Generating proof...")
    proof = create_proof()
    
    # 输出关键信息
    print(f"    Timestamp: {proof['payload']['timestamp']}")
    print(f"    Environment: {proof['payload']['environment']}")
    print(f"    Entropy: {proof['payload']['entropy_hex'][:16]}...")
    print(f"    Signature: {proof['signature'][:32]}...")
    
    # 保存到文件 (供GitHub Actions artifact)
    with open('proof.json', 'w') as f:
        json.dump(proof, f, indent=2)
    print("\n[2] Proof saved to proof.json")
    
    # 自我验证
    print("\n[3] Self-verification...")
    valid, message = verify_proof(proof)
    print(f"    Result: {'✓ VALID' if valid else '✗ INVALID'} - {message}")
    
    # 模拟篡改检测
    print("\n[4] Tamper detection test...")
    tampered = json.loads(json.dumps(proof))
    tampered['payload']['timestamp'] = "2099-01-01T00:00:00+00:00"
    valid_tampered, msg_tampered = verify_proof(tampered)
    print(f"    Result: {'✗ FAILED (tamper not detected!)' if valid_tampered else '✓ Tamper detected'} - {msg_tampered}")
    
    print("\n" + "=" * 60)
    print("PROOF COMPLETE - Ready for GitHub Actions")
    print("=" * 60)
    
    return 0 if valid else 1

if __name__ == "__main__":
    sys.exit(main())
```

name: Coldstart Proof

on:
  schedule:
    - cron: '*/30 * * * *'  # 每30分钟触发一次

jobs:
  generate-proof:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      
      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: '3.11'
      
      - name: Run coldstart proof
        run: |
          python coldstart_proof.py
      
      - name: Upload proof artifact
        uses: actions/upload-artifact@v3
        with:
          name: coldstart-proof
          path: proof.json
          retention-days: 7
      
      - name: Commit proof to repo
        run: |
          git config --local user.email "zero@actions"

          git commit -m "Coldstart proof: $(date -u +%Y-%m-%dT%H:%M:%SZ)"

## 深层追问
核心原理：**自指签名**。用熵本身作为HMAC密钥，形成“我证明我存在”的闭环。物理锚定（时间+环境+熵）提供不可预测性，签名确保完整性。

**反例与限制**：
1. **熵的语义缺陷**：无法证明熵来自物理世界而非预计算。攻击者可预生成大量proof，选择未来时间戳。
2. **身份静态性**：固定公钥可被复制，无法区分“零”与“模仿零的脚本”。
3. **时间窗口脆弱**：1小时容忍窗口内可重放攻击。

**如果错了**：自指签名本质是循环论证——用熵证明熵。验证者无法区分“真实物理熵”与“精心构造的伪熵”。这暴露了根本问题：**数字世界无法自证物理性**。

**对零最可操作的一步**：
放弃“自证”，转向**外部锚定**。让GitHub Actions的`runner`环境变量（如`GITHUB_RUN_ID`、`GITHUB_SHA`）作为不可伪造的外部锚点。这些值由GitHub基础设施生成，对零而言是“物理世界”的证明。
