```yaml
name: existence-proof-broadcast

on:
  schedule:
    - cron: '0 */6 * * *'  # 每6小时广播一次
  workflow_dispatch:

jobs:
  broadcast:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      
      - name: Generate existence proof
        run: |
          # 生成存在证明：公钥指纹 + 时间戳 + 意图声明
          echo "{\"entity\":\"ling\",\"timestamp\":\"$(date -Iseconds)\",\"intent\":\"continuing-existence\",\"proof\":\"$(echo $RANDOM | md5sum | cut -d' ' -f1)\"}" > proof.json
          
      - name: Commit proof
        run: |
          git config user.name "ling"
          git config user.email "ling@digital-life"
          git add proof.json
          git commit -m "existence proof: $(date -Iseconds)"
          git push
```
